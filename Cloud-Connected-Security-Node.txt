
#include <WiFi.h>
#include <PubSubClient.h>

#define TRIG_PIN 4
#define ECHO_PIN 18
#define BUZZER_PIN 19

const char* WIFI_SSID = "Wokwi-GUEST";
const char* WIFI_PASSWORD = "";

const char* AIO_USERNAME = "YOUR_USERNAME";
const char* AIO_KEY = "YOUR_ACTIVE_KEY";WiFiClient wifiClient;

PubSubClient mqtt(wifiClient);

void setup() {
  Serial.begin(115200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  Serial.println("Cloud Security Node Starting");

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD, 6);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi Connected!");

  mqtt.setServer("io.adafruit.com", 1883);

  while (!mqtt.connected()) {
    if (mqtt.connect("ESP32SecurityNode",
                     AIO_USERNAME, AIO_KEY)) {
      Serial.println("Adafruit IO Connected!");
    } else {
      Serial.print("MQTT failed: ");
      Serial.println(mqtt.state());
      delay(3000);
    }
  }
}

void loop() {
  if (!mqtt.connected()) {
    while (!mqtt.connected()) {
      mqtt.connect("ESP32SecurityNode",
                   AIO_USERNAME, AIO_KEY);
      delay(1000);
    }
  }

  mqtt.loop();

  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);

  if (duration > 0) {
    float distance = duration * 0.0343 / 2.0;

    Serial.print("Distance: ");
    Serial.print(distance);
    Serial.println(" cm");

    if (distance < 50) {
      tone(BUZZER_PIN, 1000);
      Serial.println("WARNING: Object Nearby!");
    } else {
      noTone(BUZZER_PIN);
      Serial.println("Area Clear");
    }

    String topic = String(AIO_USERNAME) + "/feeds/distance";
    String value = String(distance, 2);

    if (mqtt.publish(topic.c_str(), value.c_str())) {
      Serial.println("Cloud data published!");
    } else {
      Serial.println("Cloud publish failed!");
    }
  } else {
    noTone(BUZZER_PIN);
    Serial.println("No object detected");
  }

  Serial.println("------------------");
  delay(2000);
}
